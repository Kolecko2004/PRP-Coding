void pattern_idk() {
    // int m = all_values.pattern_length;
    int idy = 0;
    while (all_values.pattern[idy] != '\0') {
        fprintf(stdout, "%c", all_values.pattern[idy]);
        idy++;
    }
    fprintf(stdout, "\n");
    int m = idy;
    int lps[m];
    int j = 0, i = 1;
    
    int idx = 0;
    while (lps[idx] != '\0') {
        fprintf(stdout, "%c", lps[idx]);
        idx++;
    }
    fprintf(stdout, "\n");
    
    fprintf(stdout, "m %d\n", m);
    fprintf(stdout, "\n");
    fprintf(stdout, "j %d\n", j);
    fprintf(stdout, "i %d\n", i);

    while (i < m) {
        if (all_values.pattern[i] == all_values.pattern[j]) {
            j++;
            lps[i] = j;
            fprintf(stdout, "lps[i] %d\n", lps[i]);
            i++;
        }

        else {
            fprintf(stdout, "j %d i %d\n", j, i);
            if (j != 0) {
                j = lps[j - 1];
            }

            else {
                lps[i] = 0;
                i++;
            }
        }
    }
    all_values.lps = lps;
    idy = 0;
    while (idy < m) {
        fprintf(stdout, "%d", all_values.lps[idy]);
        idy++;
    }
    fprintf(stdout, "\n");
} 

int KMP() {
    int n = all_values.file_length;
    int m = 5;
    int j = 0, i = 0;
    while (i < n) {
        if (all_values.pattern[j] == all_values.txt_arr[i]) {
            i++;
            j++;
        }

        if (j == m) {
            fprintf(stdout, "patter found at index: %d", i-j);
            j = all_values.lps[j - 1];
        }

        else {
            if (i < n && all_values.pattern[j] != all_values.txt_arr[i]) {
                if (j != 0) {
                    j = all_values.lps[j - 1];
                }
                else {
                    i++;
                }
            }

            else {
                i++;
            }
        }
    }
}
